} } function grillSetzen() { let wert = document.getElementById("grill").value; if (wert) { document.getElementById("verbrauch").value = wert; } } function resetFlasche() { localStorage.clear(); location.reload(); } function berechnen() { let tara = parseFloat(document.getElementById("tara").value); let gesamt = parseFloat(document.getElementById("gesamt").value); let maxGas = parseFloat(document.getElementById("maxGas").value); let verbrauch = parseFloat(document.getElementById("verbrauch").value); let preis = parseFloat(document.getElementById("preis").value); if (isNaN(tara) || isNaN(gesamt) || isNaN(maxGas)) { document.getElementById("output").innerText = "Bitte alle Werte eingeben!"; return; } let rest = gesamt - tara; if (rest < 0) { document.getElementById("output").innerText = "Fehlerhafte Werte!"; return; } // Automatischer Verbrauch let lastRest = localStorage.getItem("lastRest"); let lastTime = localStorage.getItem("lastTime"); if (lastRest && lastTime) { let diffGas = lastRest - rest; let diffZeit = (Date.now() - lastTime) / 3600000; if (diffGas > 0 && diffZeit > 0) { verbrauch = diffGas / diffZeit; document.getElementById("verbrauch").value = verbrauch.toFixed(2); } } localStorage.setItem("lastRest", rest); localStorage.setItem("lastTime", Date.now()); // Startwert speichern if (!localStorage.getItem("startGas")) { localStorage.setItem("startGas", rest); } let prozent = (rest / maxGas) * 100; if (prozent > 100) { prozent = 100; } document.getElementById("output").innerText = "Restgas: " + rest.toFixed(2) + " kg (" + prozent.toFixed(1) + "%)"; // Warnung let warnText = ""; if (prozent < 10) { warnText = "🔴 FAST LEER!"; } else if (prozent < 20) { warnText = "⚠️ Bald leer"; } document.getElementById("warnung").innerText = warnText; // Countdown clearInterval(countdownInterval); if (!isNaN(verbrauch) && verbrauch > 0) { let stunden = rest / verbrauch; let tage = stunden / 24; document.getElementById("dauer").innerText = "Hält ca.: " + stunden.toFixed(1) + " h (~" + tage.toFixed(1) + " Tage)"; let sekunden = stunden * 3600; countdownInterval = setInterval(() => { if (sekunden <= 0) { document.getElementById("countdown").innerText = "Gas leer!"; clearInterval(countdownInterval); return; } let h = Math.floor(sekunden / 3600); let m = Math.floor((sekunden % 3600) / 60); let s = Math.floor(sekunden % 60); document.getElementById("countdown").innerText = "Leer in: " + h + "h " + m + "m " + s + "s"; sekunden--; }, 1000); } // Kosten let startGas = parseFloat(localStorage.getItem("startGas")); if (!isNaN(preis) && !isNaN(startGas)) { let preisProKg = preis / maxGas; let verbraucht = startGas - rest; if (verbraucht > 0) { let kosten = verbraucht * preisProKg; document.getElementById("kostenVerbraucht").innerText = "💸 Verbraucht: " + kosten.toFixed(2) + " €"; } } // Verlauf let verlauf = JSON.parse(localStorage.getItem("verlauf")) || []; verlauf.push({ zeit: Date.now(), rest: rest }); localStorage.setItem( "verlauf", JSON.stringify(verlauf) ); let verlaufText = "📊 Verlauf:

"; verlauf.slice(-5).reverse().forEach(e => { let d = new Date(e.zeit); verlaufText += d.toLocaleString() + " → " + e.rest.toFixed(2) + " kg
"; }); document.getElementById("verlauf").innerHTML = verlaufText; // Balken let bar = document.getElementById("bar"); bar.style.width = prozent + "%"; bar.className = "bar"; if (prozent > 50) { bar.classList.add("green"); } else if (prozent > 20) { bar.classList.add("yellow"); } else { bar.classList.add("red"); } }
